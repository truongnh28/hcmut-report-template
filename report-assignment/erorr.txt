function BellmanFord(start, n, flow, capacity, costo):
    distance = [Infinity] * n
    distance[start] = 0
    parent = [-1] * n
    for _ in range(n):
        for u in range(n):
            for v in range(n):            
                if flow[u][v] < capacity[u][v]:
                    if distance[u] + costo[u][v] < distance[v]:
                        distance[v] = distance[u] + costo[u][v]
                        parent[v] = u
    return distance, parent
